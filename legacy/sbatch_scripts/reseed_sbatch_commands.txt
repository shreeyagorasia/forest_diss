# ===== PINN reseed: low-weight band x both cohorts x both splits x seeds 43/44 (32 jobs) =====
for SPLIT in temporal spatial_block; do
  for COHORT in 4survey 6survey; do
    for W in 0.0 0.05 0.1 0.2; do
      for SEED in 43 44; do
        RUN_NAME="pinn_noenv_pw${W}_tw${W}_seed${SEED}"
        sbatch jobs/pinn_noenv/run_pinn_noenv.sh "${COHORT}" 500 40 "${SPLIT}" "${W}" "${W}" "${RUN_NAME}" "${SEED}"
      done
    done
  done
done
