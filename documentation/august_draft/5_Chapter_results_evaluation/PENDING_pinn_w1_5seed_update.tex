```python
import json, statistics
from pathlib import Path
from scipy import stats

root = Path("outputs")
seeds = [42,43,44,45,46]

def load_r2(split, run_name, cohort):
    p = root / split / run_name / cohort / "metrics.json"
    return json.loads(p.read_text())["r2"] if p.exists() else None

# DNN's 5-seed values are already known (Stage 4, run_name=final_dnn_seed{seed}) -- reuse, don't rerun.
for cohort in ["4survey", "6survey"]:
    for split in ["spatial_block", "temporal"]:
        dnn = [load_r2(split, f"final_dnn_seed{s}", cohort) for s in seeds]
        w1 = [load_r2(split, f"final_pinn_w1_anchorfix_seed{s}", cohort) for s in seeds]
        print(cohort, split)
        print("DNN:", dnn, "mean", statistics.mean(dnn), "sd", statistics.stdev(dnn))
        print("W1: ", w1, "mean", statistics.mean(w1), "sd", statistics.stdev(w1))
        diffs = [d-w for d,w in zip(dnn,w1)]
        print("mean_diff(DNN-W1)=", statistics.mean(diffs),
              "wilcoxon_p=", stats.wilcoxon(dnn, w1).pvalue,
              "paired_t_p=", stats.ttest_rel(dnn, w1).pvalue)
```
