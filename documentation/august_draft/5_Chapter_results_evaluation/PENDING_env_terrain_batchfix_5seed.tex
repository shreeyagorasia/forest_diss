### Step 2 — extraction after rsync

**Accuracy (straightforward, same pattern as the no-env fix):**
```python
import json, statistics
from pathlib import Path
from scipy import stats

root = Path("outputs/spatial_block")
seeds = [42,43,44,45,46]
prefixes = {"dnn_env_terrain": "dnn_env_terrain_batchfix_seed",
            "pinn_env_terrain": "pinn_env_terrain_batchfix_seed",
            "pinn_env_terrain_k": "pinnk_env_terrain_batchfix_seed"}

def load_r2(model, prefix, seed, cohort):
    p = root / f"{prefix}{seed}" / cohort / "metrics.json"
    return json.loads(p.read_text())["r2"] if p.exists() else None

for cohort in ["4survey", "6survey"]:
    results = {}
    for model, prefix in prefixes.items():
        vals = [load_r2(model, prefix, s, cohort) for s in seeds]
        results[model] = vals
        print(model, cohort, vals, "mean", statistics.mean(vals), "sd", statistics.stdev(vals))
    # pairwise: dnn vs pinn, dnn vs pinn_k, pinn vs pinn_k
    for a, b in [("dnn_env_terrain","pinn_env_terrain"), ("dnn_env_terrain","pinn_env_terrain_k"),
                 ("pinn_env_terrain","pinn_env_terrain_k")]:
        t_p = stats.ttest_rel(results[a], results[b]).pvalue
        w_p = stats.wilcoxon(results[a], results[b]).pvalue
        print(f"{a} vs {b}: paired_t_p={t_p:.4f} wilcoxon_p={w_p:.4f}")
```
